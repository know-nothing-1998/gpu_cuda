cmake_minimum_required(VERSION 3.24)

# Verified local device: RTX 4060 Laptop (compute capability 8.9).
# Override with -DCMAKE_CUDA_ARCHITECTURES=... on another device.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 89 CACHE STRING "CUDA target architectures")
endif()
project(cuda_gemm_lab VERSION 0.1.0 LANGUAGES CXX CUDA)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(CUDAToolkit 11.0 REQUIRED)

# Reconfigure on source edits so each executable identifies its exact inputs.
set(snapshot_files CMakeLists.txt src/gemm.cu src/naive.cu src/naive.cuh
    src/build_info.h.in benchmarks/matrix.csv)
set(snapshot "")
foreach(path IN LISTS snapshot_files)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" digest)
    string(APPEND snapshot "${path}:${digest}\n")
endforeach()
string(SHA256 GEMM_SOURCE_SHA256 "${snapshot}")
configure_file(src/build_info.h.in generated/build_info.h @ONLY)

add_executable(gemm src/gemm.cu src/naive.cu)
target_compile_features(gemm PRIVATE cxx_std_17 cuda_std_17)
set_target_properties(gemm PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
target_include_directories(gemm PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(gemm PRIVATE GEMM_BUILD_CONFIG="$<CONFIG>")
target_compile_options(gemm PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo;--ptxas-options=-v>)
if(MSVC)
    target_compile_options(gemm PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/W4,-utf-8>)
endif()
target_link_libraries(gemm PRIVATE CUDA::cudart CUDA::cublas)

include(CTest)
if(BUILD_TESTING)
    add_executable(harness_tests tests/harness_tests.cu src/naive.cu)
    target_compile_features(harness_tests PRIVATE cxx_std_17 cuda_std_17)
    set_target_properties(harness_tests PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
    target_include_directories(harness_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(harness_tests PRIVATE GEMM_BUILD_CONFIG="$<CONFIG>")
    if(MSVC)
        target_compile_options(harness_tests PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/W4,-utf-8>)
    endif()
    target_link_libraries(harness_tests PRIVATE CUDA::cudart CUDA::cublas)
    add_test(NAME numerical_harness COMMAND harness_tests)
    add_test(NAME cublas_correctness COMMAND gemm --mode check --kernel cublas)
    add_test(NAME cublas_unseen_boundary COMMAND gemm --mode check --kernel cublas --m 31 --n 37 --k 29)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py" $<TARGET_FILE:gemm>)
    endif()
endif()
