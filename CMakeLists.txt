# Compile one tuned specialization per build directory.
set(GEMM_TUNED_BM 64 CACHE STRING "Tuned output tile rows")
set(GEMM_TUNED_BN 64 CACHE STRING "Tuned output tile columns")
set(GEMM_TUNED_BK 64 CACHE STRING "Tuned reduction tile length")
set(GEMM_TUNED_TM 4 CACHE STRING "Tuned output rows per thread")
set(GEMM_TUNED_TN 4 CACHE STRING "Tuned output columns per thread")
set(GEMM_TUNED_PARAMETERS
    GEMM_TUNED_BM GEMM_TUNED_BN GEMM_TUNED_BK GEMM_TUNED_TM GEMM_TUNED_TN)
foreach(param IN LISTS GEMM_TUNED_PARAMETERS)
    if(NOT "${${param}}" MATCHES "^[1-9][0-9]*$")
        message(FATAL_ERROR "${param} must be a positive decimal integer; got '${${param}}'")
    endif()
    # A larger dimension cannot fit the 48 KiB FP32 tile budget below.
    # Check before multiplication so malformed inputs cannot overflow the arithmetic.
    if("${${param}}" GREATER 12288)
        message(FATAL_ERROR "${param} exceeds the 48 KiB static shared-memory tile budget")
    endif()
endforeach()
math(EXPR gemm_tuned_bm_remainder "${GEMM_TUNED_BM} % ${GEMM_TUNED_TM}")
math(EXPR gemm_tuned_bn_remainder "${GEMM_TUNED_BN} % ${GEMM_TUNED_TN}")
math(EXPR gemm_tuned_bk_vector_remainder "${GEMM_TUNED_BK} % 4")
math(EXPR gemm_tuned_bn_vector_remainder "${GEMM_TUNED_BN} % 4")
if(NOT gemm_tuned_bm_remainder EQUAL 0 OR NOT gemm_tuned_bn_remainder EQUAL 0)
    message(FATAL_ERROR "Tuned tiles require BM % TM == 0 and BN % TN == 0")
endif()
if(NOT gemm_tuned_bk_vector_remainder EQUAL 0 OR NOT gemm_tuned_bn_vector_remainder EQUAL 0)
    message(FATAL_ERROR "Tuned float4 loading requires BK and BN to be multiples of 4")
endif()
math(EXPR gemm_tuned_threads
    "(${GEMM_TUNED_BM} / ${GEMM_TUNED_TM}) * (${GEMM_TUNED_BN} / ${GEMM_TUNED_TN})")
if(gemm_tuned_threads GREATER 1024)
    message(FATAL_ERROR "Tuned configuration needs ${gemm_tuned_threads} threads; maximum is 1024")
endif()
math(EXPR gemm_tuned_shared_bytes
    "4 * ${GEMM_TUNED_BK} * (${GEMM_TUNED_BM} + ${GEMM_TUNED_BN})")
if(gemm_tuned_shared_bytes GREATER 49152)
    message(FATAL_ERROR
        "Tuned configuration needs ${gemm_tuned_shared_bytes} static shared bytes; maximum is 49152")
endif()

# Separate experimental warp-tiling specialization; the tuned kernel is unchanged.
set(GEMM_WARP_BM 64 CACHE STRING "Warp-tiled block rows")
set(GEMM_WARP_BN 64 CACHE STRING "Warp-tiled block columns")
set(GEMM_WARP_BK 16 CACHE STRING "Warp-tiled reduction tile")
set(GEMM_WARP_WM 32 CACHE STRING "Output rows per warp")
set(GEMM_WARP_WN 32 CACHE STRING "Output columns per warp")
set(GEMM_WARP_TM 4 CACHE STRING "Thread subtile rows")
set(GEMM_WARP_TN 4 CACHE STRING "Thread subtile columns")
set(GEMM_WARP_WNITER 1 CACHE STRING "Warp subtiles along N")
option(GEMM_WARP_RUNTIME_CONFIGS "Precompile the warp search-space configurations for CLI selection" ON)
# One warp per WM x WN tile; thread count is derived, not a tuning input.
unset(GEMM_WARP_THREADS CACHE)
set(GEMM_WARP_PARAMETERS GEMM_WARP_BM GEMM_WARP_BN GEMM_WARP_BK
    GEMM_WARP_WM GEMM_WARP_WN GEMM_WARP_TM GEMM_WARP_TN
    GEMM_WARP_WNITER)
function(gemm_validate_warp_config label BM BN BK WM WN TM TN WNITER)
    foreach(param BM BN BK WM WN TM TN WNITER)
        if(NOT "${${param}}" MATCHES "^[1-9][0-9]*$" OR "${${param}}" GREATER 12288)
            message(FATAL_ERROR "${label}: ${param} must be a positive integer <= 12288")
        endif()
    endforeach()
    math(EXPR bm_rem "${BM} % ${WM}")
    math(EXPR bn_rem "${BN} % ${WN}")
    math(EXPR bk_rem "${BK} % 4")
    math(EXPR bn_vec_rem "${BN} % 4")
    math(EXPR threads "32 * (${BM} / ${WM}) * (${BN} / ${WN})")
    math(EXPR shared_bytes "4 * ${BK} * (${BM} + ${BN})")
    if(NOT bm_rem EQUAL 0 OR NOT bn_rem EQUAL 0
       OR NOT bk_rem EQUAL 0 OR NOT bn_vec_rem EQUAL 0
       OR threads GREATER 1024 OR shared_bytes GREATER 49152)
        message(FATAL_ERROR "${label}: invalid warp coverage, vector width, thread count or shared memory budget")
    endif()
    math(EXPR sub_area "32 * ${TM} * ${TN} * ${WNITER}")
    math(EXPR warp_area "${WM} * ${WN}")
    math(EXPR area_rem "${warp_area} % ${sub_area}")
    math(EXPR m_iters "${warp_area} / ${sub_area}")
    if(NOT area_rem EQUAL 0 OR m_iters LESS 1)
        message(FATAL_ERROR "${label}: warp tile must contain an integral positive number of thread subtiles")
    endif()
    math(EXPR wm_rem "${WM} % ${m_iters}")
    math(EXPR wn_rem "${WN} % ${WNITER}")
    math(EXPR subm_rem "(${WM} / ${m_iters}) % ${TM}")
    math(EXPR subn_rem "(${WN} / ${WNITER}) % ${TN}")
    if(NOT wm_rem EQUAL 0 OR NOT wn_rem EQUAL 0
       OR NOT subm_rem EQUAL 0 OR NOT subn_rem EQUAL 0)
        message(FATAL_ERROR "${label}: warp subtiles must divide evenly among lanes")
    endif()
endfunction()

set(warp_default_values "")
foreach(param IN LISTS GEMM_WARP_PARAMETERS)
    list(APPEND warp_default_values "${${param}}")
endforeach()
gemm_validate_warp_config("Default warptiled configuration" ${warp_default_values})
list(JOIN warp_default_values ", " warp_default_tuple)
set(warp_runtime_tuples "${warp_default_tuple}")

# Validate the catalog even in single-configuration builds; only instantiation is optional.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/benchmarks/warptiling.json" warp_space_json)
set(warp_parameter_names BM BN BK WM WN TM TN WNITER)
string(JSON warp_order_count LENGTH "${warp_space_json}" parameter_order)
if(NOT warp_order_count EQUAL 8)
    message(FATAL_ERROR "benchmarks/warptiling.json: parameter_order must contain eight names")
endif()
foreach(param_index RANGE 0 7)
    string(JSON actual_name GET "${warp_space_json}" parameter_order ${param_index})
    list(GET warp_parameter_names ${param_index} expected_name)
    if(NOT actual_name STREQUAL expected_name)
        message(FATAL_ERROR "benchmarks/warptiling.json: unexpected parameter_order entry ${param_index}")
    endif()
endforeach()
string(JSON warp_candidate_count LENGTH "${warp_space_json}" candidates)
if(warp_candidate_count LESS 1)
    message(FATAL_ERROR "benchmarks/warptiling.json: candidates must not be empty")
endif()
math(EXPR warp_candidate_last "${warp_candidate_count} - 1")
foreach(candidate_index RANGE 0 ${warp_candidate_last})
    string(JSON candidate_count LENGTH "${warp_space_json}" candidates ${candidate_index})
    if(NOT candidate_count EQUAL 8)
        message(FATAL_ERROR "benchmarks/warptiling.json: candidate ${candidate_index} must contain eight parameters")
    endif()
    set(candidate_values "")
    foreach(param_index RANGE 0 7)
        string(JSON value_type TYPE "${warp_space_json}" candidates ${candidate_index} ${param_index})
        if(NOT value_type STREQUAL "NUMBER")
            message(FATAL_ERROR "benchmarks/warptiling.json: candidate ${candidate_index} parameters must be numbers")
        endif()
        string(JSON value GET "${warp_space_json}" candidates ${candidate_index} ${param_index})
        list(APPEND candidate_values "${value}")
    endforeach()
    gemm_validate_warp_config("benchmarks/warptiling.json candidate ${candidate_index}" ${candidate_values})
    if(GEMM_WARP_RUNTIME_CONFIGS)
        list(JOIN candidate_values ", " candidate_tuple)
        list(APPEND warp_runtime_tuples "${candidate_tuple}")
    endif()
endforeach()
list(REMOVE_DUPLICATES warp_runtime_tuples)
set(GEMM_WARP_VARIANT_ENTRIES "")
foreach(variant_tuple IN LISTS warp_runtime_tuples)
    string(APPEND GEMM_WARP_VARIANT_ENTRIES "GEMM_WARPTILED_VARIANT(${variant_tuple})\n")
endforeach()

# Reconfigure on source edits so each executable identifies its exact inputs.
set(snapshot_files CMakeLists.txt src/gemm.cu
    ${GEMM_KERNEL_SOURCES} ${GEMM_KERNEL_HEADERS}
    src/build_info.h.in src/tuned_config.h.in src/warptiled_config.h.in
    src/warptiled_variants.inc.in benchmarks/matrix.csv benchmarks/warptiling.json)
set(snapshot "")
foreach(path IN LISTS snapshot_files)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" digest)
    string(APPEND snapshot "${path}:${digest}\n")
endforeach()
foreach(param IN LISTS GEMM_TUNED_PARAMETERS GEMM_WARP_PARAMETERS)
    string(APPEND snapshot "${param}:${${param}}\n")
endforeach()
string(APPEND snapshot "GEMM_WARP_RUNTIME_CONFIGS:${GEMM_WARP_RUNTIME_CONFIGS}\n")
string(SHA256 GEMM_SOURCE_SHA256 "${snapshot}")
configure_file(src/build_info.h.in generated/build_info.h @ONLY)
configure_file(src/tuned_config.h.in generated/tuned_config.h @ONLY)
configure_file(src/warptiled_config.h.in generated/warptiled_config.h @ONLY)
configure_file(src/warptiled_variants.inc.in generated/warptiled_variants.inc @ONLY)
